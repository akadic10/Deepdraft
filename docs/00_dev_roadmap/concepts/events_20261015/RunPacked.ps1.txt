param([Parameter(Mandatory=$true)][string]$Package)
$ErrorActionPreference='Stop'
$smokeRoot='P:/Deepdraft/tmp/events_20261015/packed'
New-Item -ItemType Directory -Force $smokeRoot | Out-Null
$profile=$smokeRoot+'/profile'
$saveDir=$profile+'/Godot/app_userdata/Deepdraft/saves'
New-Item -ItemType Directory -Force $saveDir | Out-Null
$source='P:/Deepdraft/tmp/events_20261015/input_quicksave.json'
$sourceHash=(Get-FileHash -LiteralPath $source -Algorithm SHA256).Hash
Copy-Item -LiteralPath $source -Destination ($saveDir+'/quicksave.json')
$oldAppData=$env:APPDATA
$oldLocalAppData=$env:LOCALAPPDATA
try {
    $env:APPDATA=$profile
    $env:LOCALAPPDATA=$profile
    $args=@('--path',('"'+$Package+'"'),'--main-pack',('"'+$Package+'/Deepdraft.pck"'),'--audio-driver','Dummy','--script','P:/Deepdraft/tools/performance/SmokeEventsExport.gd','--',('--output='+$smokeRoot))
    $p=Start-Process -FilePath 'S:/STEAM/steamapps/common/Godot Engine/godot.windows.opt.tools.64.exe' -ArgumentList $args -WorkingDirectory $Package -WindowStyle Hidden -PassThru -RedirectStandardOutput ($smokeRoot+'/run.out') -RedirectStandardError ($smokeRoot+'/run.err')
    $null=$p.Handle
    if(-not $p.WaitForExit(150000)){Stop-Process -Id $p.Id; throw 'Packed smoke timeout'}
    Get-Content ($smokeRoot+'/run.out') -Tail 3
    if($p.ExitCode -ne 0 -or (Select-String -LiteralPath ($smokeRoot+'/run.err') -Pattern '^SCRIPT ERROR:|^ERROR:' -Quiet)){Get-Content ($smokeRoot+'/run.err') -Tail 35; throw 'Packed check failed'}
    if(-not (Select-String -LiteralPath ($smokeRoot+'/run.out') -Pattern '^EVENTS_EXPORT_PASS' -Quiet)){throw 'Missing packed pass marker'}
    if((Get-FileHash -LiteralPath $source -Algorithm SHA256).Hash -ne $sourceHash){throw 'Source copy changed'}
    $env:APPDATA=$smokeRoot+'/release_profile'
    $env:LOCALAPPDATA=$env:APPDATA
    New-Item -ItemType Directory -Force $env:APPDATA | Out-Null
    $p=Start-Process -FilePath ($Package+'/Deepdraft.exe') -ArgumentList @('--audio-driver','Dummy','--quit-after','1800') -WorkingDirectory $Package -WindowStyle Hidden -PassThru -RedirectStandardOutput ($smokeRoot+'/release.out') -RedirectStandardError ($smokeRoot+'/release.err')
    $null=$p.Handle
    if(-not $p.WaitForExit(120000)){Stop-Process -Id $p.Id; throw 'Release startup timeout'}
    if($p.ExitCode -ne 0 -or (Select-String -LiteralPath ($smokeRoot+'/release.err') -Pattern '^SCRIPT ERROR:|^ERROR:' -Quiet)){throw 'Release startup errors'}
    if(-not (Select-String -LiteralPath ($smokeRoot+'/release.out') -Pattern 'built block-face overview' -Quiet)){throw 'Release did not generate world'}
    @{release_startup_pass=$true;exit_code=$p.ExitCode;package=$Package;source_save_sha256=$sourceHash;source_save_unchanged=$true}|ConvertTo-Json|Set-Content ($smokeRoot+'/release_startup.json')
    Write-Output 'RELEASE_STARTUP_PASS'
} finally {$env:APPDATA=$oldAppData; $env:LOCALAPPDATA=$oldLocalAppData}
