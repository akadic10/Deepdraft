from pathlib import Path
r=Path('P:/Deepdraft')
p=r/'docs/00_dev_roadmap/concepts/lighting_20261015/UndergroundLighting.gd.before.txt'
s=p.read_text(encoding='utf-8')
a=s.index('func _process(delta: float) -> void:')
b=s.index('\n\n## Actors',a)
s=s[:a]+'''var trace_totals := {}
var trace_frames := 0
var trace_uploads: Array = []

func _process(delta: float) -> void:
	var frame_start := Time.get_ticks_usec()
	var before_phase := _phase
	var counts := {"air":_air.size(),"visited":_visited.size(),"pending":_pending.size(),"slots":_slots.size()}
	var start := frame_start
	_fog_elapsed += delta
	if _fog_elapsed >= 0.05:
		_fog_elapsed = 0.0
		_sync_fog()
	var timings := {"fog":Time.get_ticks_usec()-start}
	start = Time.get_ticks_usec()
	var worst := 0
	var worst_phase := ""
	while is_updating() and Time.get_ticks_usec() - start < FRAME_BUDGET_USEC:
		var phase := _phase
		var step_start := Time.get_ticks_usec()
		_step()
		var elapsed := Time.get_ticks_usec()-step_start
		if elapsed>worst:
			worst=elapsed
			worst_phase=phase
	timings["solve"]=Time.get_ticks_usec()-start
	start = Time.get_ticks_usec()
	trace_uploads.clear()
	_upload()
	timings["upload"]=Time.get_ticks_usec()-start
	start = Time.get_ticks_usec()
	_sync_actors()
	timings["actors"]=Time.get_ticks_usec()-start
	timings["total"]=Time.get_ticks_usec()-frame_start
	trace_frames+=1
	for key: String in timings:
		var entry: Dictionary=trace_totals.get(key,{"sum":0,"max":0})
		entry.sum+=int(timings[key])
		entry.max=maxi(entry.max,timings[key])
		trace_totals[key]=entry
	if int(timings.total)>10000:
		print("LIGHTING_SPIKE ",JSON.stringify({"usec":timings,"phase":before_phase,"worst_step_usec":worst,"worst_step_phase":worst_phase,"before_counts":counts,"uploads":trace_uploads}))
''' + s[b:]
s=s.replace('\t\t\t_pages[page].create_from_images(images)', '''\t\t\tvar upload_start:=Time.get_ticks_usec()
			_pages[page].create_from_images(images)
			trace_uploads.append({"page":page,"layers":images.size(),"kind":"create","usec":Time.get_ticks_usec()-upload_start})''')
s=s.replace('\t\t\t_pages[page].update_layer(_images[page][slot%PAGE_LAYERS],slot%PAGE_LAYERS)', '''\t\t\tvar upload_start:=Time.get_ticks_usec()
			_pages[page].update_layer(_images[page][slot%PAGE_LAYERS],slot%PAGE_LAYERS)
			trace_uploads.append({"page":page,"kind":"update","usec":Time.get_ticks_usec()-upload_start})''')
(p.parent/'UndergroundLighting.trace.txt').write_text(s,encoding='utf-8')
print('Diagnostic source prepared; not installed.')
