extends SceneTree
var done := false
var loaded := false

func _init() -> void: _run.call_deferred()

func _run() -> void:
	if not "/dwarf_concerns_20261017/" in OS.get_user_data_dir().replace("\\","/"): quit(2); return
	create_timer(240).timeout.connect(func(): push_error("PLAYER_NEEDS_TIMEOUT"); quit(1))
	var saves = root.get_node("SaveManager")
	var clock_node = root.get_node("WorldClock")
	saves.configure_storage_for_testing("user://isolated")
	DirAccess.make_dir_recursive_absolute("user://isolated")
	DirAccess.copy_absolute("res://tmp/hauling_save_20261017/source/quicksave.json","user://isolated/quicksave.json")
	saves.set_process(false)
	change_scene_to_file("res://scenes/main/debug_world.tscn")
	for i in 12: await process_frame
	if current_scene.get_node("DwarfNeedIndicators").visible:
		push_error("Indicators visible during loading"); quit(1); return
	saves.set_process(true)
	saves.load_finished.connect(func(ok: bool,_backup: bool): done=true; loaded=ok)
	if not saves.request_load(): push_error("LOAD_REJECTED"); quit(1); return
	while not done: await process_frame
	if not loaded: push_error("LOAD_FAILED"); quit(1); return
	clock_node.set_paused(true)
	while current_scene.get_node("WorldLoadingScreen").visible: await process_frame
	saves.set_process(false)
	var director = current_scene.get_node("DwarfDirector")
	var indicators = current_scene.get_node("DwarfNeedIndicators")
	var camera = current_scene.get_node("CameraRig")
	var target = root.get_node("TaskManager")._agents[1]
	camera.locate_subject(target)
	camera.global_position=camera._target_pos
	camera._target_zoom=36.0
	camera.spring_arm.spring_length=36.0
	director.inspect_dwarf(target)
	for i in 60: await process_frame
	indicators.refresh()
	var report := {"loaded":loaded,"loading_hidden":true,"states":[],"visible_badges":0}
	var concern = load("res://scripts/components/DwarfConcern.gd")
	for agent in director.get_roster():
		report.states.append({"name":agent.dwarf_name,"food":agent._food.value,"concern":concern.describe(agent)})
	for entry in indicators._entries.values():
		if entry.icon.visible: report.visible_badges+=1
	if report.visible_badges<1: push_error("No visible badges in copied save"); quit(1); return
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://tmp/dwarf_concerns_20261017/player_save.png")
	var out=FileAccess.open("res://tmp/dwarf_concerns_20261017/player_report.json",FileAccess.WRITE)
	out.store_string(JSON.stringify(report,"\t")); out.close()
	print("PLAYER_NEED_INDICATORS_OK ",JSON.stringify(report))
	quit(0)
