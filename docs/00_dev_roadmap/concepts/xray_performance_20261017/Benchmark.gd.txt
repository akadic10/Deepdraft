extends SceneTree
const Detail = preload("res://tools/performance/RenderProfile.gd")
var loaded := false
var report := {}
var xray
func _init() -> void:
	_run.call_deferred()
func _run() -> void:
	if not "/xray_performance_20261017/" in OS.get_user_data_dir().replace("\\","/"): quit(2); return
	create_timer(300).timeout.connect(func(): push_error("XRAY_BENCH_TIMEOUT"); quit(1))
	var saves=root.get_node("SaveManager")
	var clock_node=root.get_node("WorldClock")
	saves.configure_storage_for_testing("user://isolated")
	DirAccess.make_dir_recursive_absolute("user://isolated")
	DirAccess.copy_absolute("res://tmp/xray_performance_20261017/source/quicksave.json","user://isolated/quicksave.json")
	var placeholder=Node3D.new()
	placeholder.scene_file_path="res://scenes/main/debug_world.tscn"
	root.add_child(placeholder); current_scene=placeholder
	saves.load_finished.connect(func(ok: bool,_backup: bool): loaded=true; assert(ok))
	assert(saves.request_load())
	while not loaded: await process_frame
	clock_node.set_paused(true); saves.set_process(false)
	while current_scene.get_node("WorldLoadingScreen").visible: await process_frame
	DisplayServer.window_set_vsync_mode(DisplayServer.VSYNC_DISABLED)
	Engine.max_fps=0
	var rig=current_scene.get_node("CameraRig")
	rig.set_process(false)
	xray=current_scene.get_node("XrayController")
	var slice=current_scene.get_node("SliceController")
	slice.show_at_height(47); xray.pin_reference(); slice.show_at_height(53)
	RenderingServer.viewport_set_measure_render_time(root.get_viewport_rid(),true)
	report={"gpu":RenderingServer.get_video_adapter_name(),"cpu":OS.get_processor_name(),"resolution":str(root.size),"camera":str(rig.global_transform),"phases":{}}
	await _sample("off")
	xray.set_mode("full")
	await _sample("full_start")
	await _sample("full_steady")
	xray.set_process(false)
	await _sample("full_frozen_controller")
	xray.set_process(true)
	clock_node.set_paused(false)
	await _sample("full_active")
	clock_node.set_paused(true)
	FileAccess.open("user://report.json",FileAccess.WRITE).store_string(JSON.stringify(report,"\t"))
	print("XRAY_BENCH_OK")
	quit(0)
func _sample(label: String) -> void:
	for i in 20: await process_frame
	var rows=[]
	Detail.take()
	var start=Time.get_ticks_usec()
	var previous=start
	while Time.get_ticks_usec()-start<12000000:
		await process_frame
		var now=Time.get_ticks_usec()
		rows.append({"frame":(now-previous)/1000.0,"gpu":RenderingServer.viewport_get_measured_render_time_gpu(root.get_viewport_rid()),"render_cpu":RenderingServer.viewport_get_measured_render_time_cpu(root.get_viewport_rid()),"detail":Detail.take()})
		previous=now
	var summary={"frames":rows.size(),"scan":xray._scan.size(),"visuals":xray._visuals.size(),"moving":xray._moving.size(),"dirty":xray._dirty.size(),"chunks":xray._chunks.size(),"detail":{}}
	for key in ["frame","gpu","render_cpu"]:
		var values=[]
		for row in rows: values.append(row[key])
		summary[key]=_stats(values)
	var keys={}
	for row in rows:
		for key in row.detail: keys[key]=true
	for key in keys:
		var values=[]
		for row in rows: values.append(row.detail.get(key,0))
		summary.detail[key]=_stats(values)
	report.phases[label]=summary
	FileAccess.open("user://"+label+".json",FileAccess.WRITE).store_string(JSON.stringify(rows))
	FileAccess.open("user://report.json",FileAccess.WRITE).store_string(JSON.stringify(report,"\t"))
	print("XRAY_SAMPLE ",label," ",JSON.stringify(summary))
	if label=="full_steady":
		await RenderingServer.frame_post_draw
		root.get_texture().get_image().save_png("user://full.png")
func _stats(values: Array) -> Dictionary:
	values.sort()
	return {"median":values[values.size()/2],"p95":values[int(values.size()*.95)],"max":values.back()}
