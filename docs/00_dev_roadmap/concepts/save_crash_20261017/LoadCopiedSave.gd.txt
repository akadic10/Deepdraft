extends SceneTree

var saves
var finished := false
var succeeded := false
var autosave := false

func _init() -> void:
	_run.call_deferred()

func _run() -> void:
	if not "/save_crash_20261017/" in OS.get_user_data_dir().replace("\\","/"): quit(2); return
	create_timer(150).timeout.connect(func(): push_error("Copied-save load timed out"); quit(1))
	saves = root.get_node("SaveManager")
	autosave = "--autosave" in OS.get_cmdline_user_args()
	var file_name := "autosave.json" if autosave else "quicksave.json"
	var source := "res://tmp/save_crash_20261017/source/"+file_name
	var expected: Dictionary = JSON.parse_string(FileAccess.get_file_as_string(source)).scene
	saves.configure_storage_for_testing("user://save_copy")
	DirAccess.make_dir_recursive_absolute("user://save_copy")
	DirAccess.copy_absolute(source,"user://save_copy/"+file_name)
	saves.set_process(false)
	root.get_node("WorldClock").set_paused(true)
	change_scene_to_file("res://scenes/main/debug_world.tscn")
	await process_frame
	while current_scene==null: await process_frame
	while current_scene.get_node("WorldLoadingScreen").visible: await process_frame
	print("COPIED_SAVE_NEW_WORLD_READY")
	saves.set_process(true)
	saves.load_finished.connect(func(ok: bool,_backup: bool): finished=true; succeeded=ok)
	var accepted: bool = saves.request_load_autosave() if autosave else saves.request_load()
	if not accepted: push_error("Copied save rejected"); quit(1); return
	while not finished: await process_frame
	if not succeeded: quit(1); return
	saves.set_process(false)
	root.get_node("WorldClock").set_paused(true)
	while current_scene.get_node("WorldLoadingScreen").visible: await process_frame
	print("COPIED_SAVE_LOADED")
	var mining = current_scene.get_node("MiningDesignationController")
	var construction = current_scene.get_node("ConstructionController")
	var missing := 0
	for entry in expected.mining.mined_blocks:
		if not mining._mined_blocks.has(saves.unpack_v3i(entry)): missing += 1
	for entry in expected.construction.built:
		var cell: Vector3i = saves.unpack_v3i(entry.cell)
		if construction.built.get(cell,"")!=entry.block: missing += 1
		if root.get_node("WorldData").get_terrain_block(cell.x,cell.y,cell.z)!=root.get_node("BlockRegistry").get_id(entry.block): missing += 1
	if missing>0 or mining._mined_blocks.size()!=expected.mining.mined_blocks.size() or construction.built.size()!=expected.construction.built.size():
		push_error("Restored terrain differs from save"); quit(1); return
	print("COPIED_SAVE_TERRAIN_EXACT mined=",mining._mined_blocks.size()," built=",construction.built.size())
	await create_timer(5).timeout
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://tmp/save_crash_20261017/loaded_"+file_name+".png")
	var xray = current_scene.get_node("XrayController")
	xray.set_mode("full")
	while not xray._dirty.is_empty() or not xray._job.is_empty() or xray._reference_dirty or not xray._scan.is_empty(): await process_frame
	await create_timer(1).timeout
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png("res://tmp/save_crash_20261017/xray_"+file_name+".png")
	xray.set_mode("off")
	print("COPIED_SAVE_LOAD_OK")
	quit(0)
