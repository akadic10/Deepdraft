extends SceneTree
var saves
var tasks
var stocks
var nav
var done := false
var result := {}
func _init() -> void: _run.call_deferred()
func _run() -> void:
	if not "/hauling_save_20261017/" in OS.get_user_data_dir().replace("\\","/"): quit(2); return
	create_timer(360).timeout.connect(func(): push_error("HAUL_DIAGNOSTIC_TIMEOUT"); quit(1))
	saves=root.get_node("SaveManager"); tasks=root.get_node("TaskManager"); stocks=root.get_node("StockpileManager"); nav=root.get_node("NavGrid")
	saves.configure_storage_for_testing("user://isolated")
	DirAccess.make_dir_recursive_absolute("user://isolated")
	DirAccess.copy_absolute("res://tmp/hauling_save_20261017/source/quicksave.json","user://isolated/quicksave.json")
	saves.set_process(false)
	root.get_node("WorldClock").set_paused(true)
	change_scene_to_file("res://scenes/main/debug_world.tscn")
	while current_scene==null: await process_frame
	for i in 12: await process_frame
	saves.set_process(true)
	saves.load_finished.connect(func(ok: bool,_backup: bool): done=true; if not ok: push_error("LOAD_FAILED"))
	if not saves.request_load(): push_error("LOAD_REJECTED"); quit(1); return
	while not done: await process_frame
	root.get_node("WorldClock").set_paused(true)
	while current_scene.get_node("WorldLoadingScreen").visible: await process_frame
	saves.set_process(false)
	result.initial=_report()
	result.routes=[]
	for worker in tasks._agents.values():
		for zone in stocks._zones.values():
			var target: Vector3i=zone.nearest_stand_target(worker.current_cell())
			var path: Array=nav.find_path(worker.current_cell(),target,10000)
			var quote={}
			zone.advance_haul_quote(worker.current_cell(),quote,Time.get_ticks_usec()+1000000)
			result.routes.append({"worker":worker.dwarf_name,"from":str(worker.current_cell()),"zone":zone.zone_id,
				"target":str(target),"walkable":nav.is_walkable(target),"path_size":path.size(),"quote_cell":str(quote.get("cell")),
				"room":zone._has_any_room(),"leases":zone._lease_ids.keys(),"drop_manager":is_instance_valid(zone.drop_manager)})
	_write()
	print("HAUL_INITIAL ",JSON.stringify(result))
	var drops=get_first_node_in_group("item_drop_manager")
	var storage=load("res://scripts/components/StorageComponent.gd")
	result.pickups=[]
	var worker=tasks._agents[1]
	for node in drops._loose:
		var cell=drops.item_floor_cell(node)
		var goals={}
		for stand in storage.ground_access_cells(cell):
			if nav.is_walkable(stand): goals[stand]=true
		var query={}
		var route_status=nav.advance_reachability(query,worker.current_cell(),cell,24000,0,false,goals) if not goals.is_empty() else -1
		while route_status==nav.ProbeResult.SEARCHING:
			route_status=nav.advance_reachability(query,worker.current_cell(),cell,24000,0,false,goals)
		result.pickups.append({"key":drops._loose[node],"cell":str(cell),"stands":str(goals.keys()),"route_status":route_status,"expanded":query.get("expanded",0),"path":str(query.get("path",[]))})
	_write()
	print("HAUL_PICKUPS ",JSON.stringify(result.pickups))
	root.get_node("WorldClock").set_paused(false)
	result.samples=[]
	for i in (12 if "--verify" in OS.get_cmdline_user_args() else 6):
		await create_timer(10).timeout
		var sample=_report(); result.samples.append(sample)
		print("HAUL_SAMPLE ",i," ",JSON.stringify(sample))
		_write()
	if "--verify" in OS.get_cmdline_user_args():
		var final: Dictionary=result.samples.back()
		if final.stock.stored<=result.initial.stock.stored or final.total_units!=result.initial.total_units:
			push_error("PLAYER_HAUL_DELIVERY_OR_CONSERVATION_FAILED"); quit(1); return
		if not tasks.lifecycle_errors().is_empty(): push_error(str(tasks.lifecycle_errors())); quit(1); return
		print("PLAYER_HAUL_DELIVERY_OK stored=",result.initial.stock.stored,"->",final.stock.stored," units=",final.total_units)
	print("HAUL_DIAGNOSTIC_OK")
	quit(0)
func _report() -> Dictionary:
	var r={"idle":tasks._idle_dwarves.duplicate(),"stock":stocks.get_stats(),"workers":[],"tasks":[],"nav_revision":nav.navigation_revision,"stock_revision":stocks.hauling_revision,"wake_count":tasks._wake_count,"frontiers":[]}
	var drops=get_first_node_in_group("item_drop_manager")
	r.total_units=int(r.stock.stored)
	for node in drops._loose: r.total_units+=drops.quantity_of(node)
	for id in tasks._family_frontiers:
		var ctx=tasks._family_frontiers[id]
		r.frontiers.append({"task":id,"index":ctx.get("index"),"probe":ctx.get("probe_index"),"expanded":ctx.get("route_query",{}).get("expanded"),"candidates":ctx.get("candidates",[]).size(),"excluded":str(ctx.get("excluded",{})),"revision":ctx.get("revision"),"idle_version":ctx.get("idle_version"),"current_idle_version":tasks._idle_version})
	for worker in tasks._agents.values():
		for entry in worker._carried_entries: r.total_units+=drops.quantity_of(entry[0])
		r.workers.append({"id":worker.dwarf_id,"name":worker.dwarf_name,"cell":str(worker.current_cell()),"walkable":nav.is_walkable(worker.current_cell()),
			"task":worker.current_task_id,"phase":worker._task_phase,"food":worker._food.value,"food_state":worker._food.state,"food_message":worker._food.message,
			"sleeping":worker.is_sleeping(),"cargo":worker._carried_entries.size()})
	for task in tasks._tasks.values():
		r.tasks.append({"id":task.id,"type":task.type,"status":task.status,"worker":task.assigned_to,"source":task.source_id,"wait":task.wait_reason,"blocked":task.blocked_count})
	return r
func _write() -> void:
	var f=FileAccess.open("res://tmp/hauling_save_20261017/diagnostic.json",FileAccess.WRITE)
	f.store_string(JSON.stringify(result,"\t")); f.close()
