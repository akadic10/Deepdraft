"""Install reversible diagnostic wrappers, restoring exact bytes after the run.

Never instrument worker methods. Snapshot + SHA guards preserve pre-existing work
and refuse to overwrite concurrent edits. RunSteadyRendering.ps1 owns the lifetime.
"""
import argparse
import hashlib
import json
import re
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
TARGETS = {
    "scripts/systems/WorldRenderer.gd": [
        "_process", "_update_block_face_overview", "_drain_overview_tile_queue",
        "_recompute_overview_stats", "_rebuild_cavity_shell", "_rebuild_added_terrain",
    ],
    "scripts/components/UndergroundLighting.gd": [
        "_process", "_sync_actors", "actor_sky_at", "_sync_fog", "_upload",
    ],
    "scripts/entities/DwarfAgent.gd": ["_process"],
}


def sha(data):
    return hashlib.sha256(data).hexdigest()


def instrument(text, source, methods, fog_control=False):
    helper = 'preload("res://tools/performance/RenderProfile.gd")'
    for method in methods:
        pattern = rf"^func {method}\(([^\n]*)\) -> (\w+):$"
        match = re.search(pattern, text, re.M)
        if not match:
            raise ValueError(f"Missing simple signature: {source}:{method}")
        args, result = match.groups()
        callargs = ", ".join(a.strip().split(":")[0] for a in args.split(",") if a.strip())
        renamed = "_render_profile_original" + method
        call = f"{renamed}({callargs})"
        call_line = call if result == "void" else f"var profile_result: {result} = {call}"
        wrapper = (
            f"func {method}({args}) -> {result}:\n"
            "\tvar profile_started := Time.get_ticks_usec()\n"
            f"\t{call_line}\n"
            f'\t{helper}.record("{Path(source).stem}/{method}", profile_started)\n'
        )
        if fog_control and source.endswith("UndergroundLighting.gd") and method == "_sync_fog":
            header, body = wrapper.split("\n", 1)
            wrapper = header + '\n\tif preload("res://tools/performance/FogBroadcastSwitch.gd").disabled: return\n' + body
        if result != "void":
            wrapper += "\treturn profile_result\n"
        wrapper += f"\nfunc {renamed}({args}) -> {result}:"
        text = text[:match.start()] + wrapper + text[match.end():]
    if source.endswith("UndergroundLighting.gd"):
        # Time the existing whole solver budget loop; per-cell timing would distort it.
        needle = "\t\t_step()\n\t_upload()"
        assert text.count(needle) == 1
        text = text.replace(needle, '\t\t_step()\n\t' + helper + '.record("UndergroundLighting/solver_loop", start)\n\t_upload()')
        needle = '\t\tfor material: ShaderMaterial in _material_cache.values(): material.set_shader_parameter(key, values[key])'
        if needle in text:
            assert text.count(needle) == 1
            text = text.replace(needle, '\t\t' + helper + '.count("fog/material_parameter_writes", _material_cache.size())\n' + needle)
        else:
            needle = '\t\t_atmosphere.update(_fog_parameters)'
            assert text.count(needle) == 1
            text = text.replace(needle, '\t\t' + helper + '.count("fog/shared_texture_updates", 1)\n' + needle)
        needle = '\t\t\tmesh.set_instance_shader_parameter("actor_sky_access", amount)'
        assert text.count(needle) == 1
        text = text.replace(needle, '\t\t\t' + helper + '.count("actors/changed_instance_writes", 1)\n' + needle)
    return text


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("mode", choices=["install", "restore"])
    parser.add_argument("folder", type=Path)
    parser.add_argument("--fog-control", action="store_true")
    args = parser.parse_args()
    folder = args.folder.resolve()
    if not folder.is_relative_to(ROOT / "tmp"):
        raise ValueError("Diagnostic backups must be within workspace tmp")
    manifest_path = folder / "instrumentation.json"
    if args.mode == "install":
        if manifest_path.exists():
            raise ValueError("Refusing to overwrite an existing diagnostic snapshot")
        folder.mkdir(parents=True, exist_ok=True)
        manifest = {}
        pending = {}
        for source, methods in TARGETS.items():
            original = (ROOT / source).read_bytes()
            modified = instrument(original.decode("utf-8").replace("\r\n", "\n"), source, methods, args.fog_control).encode("utf-8")
            filename = Path(source).name + ".before.txt"
            (folder / filename).write_bytes(original)
            (folder / (Path(source).name + ".instrumented.txt")).write_bytes(modified)
            manifest[source] = {"before":sha(original), "instrumented":sha(modified), "backup":filename}
            pending[source] = modified
        manifest_path.write_text(json.dumps(manifest, indent=2), encoding="utf-8")
        for source, modified in pending.items():
            if sha((ROOT / source).read_bytes()) != manifest[source]["before"]:
                raise ValueError(f"Concurrent source edit: {source}")
            (ROOT / source).write_bytes(modified)
    else:
        manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
        for source, data in manifest.items():
            if sha((ROOT / source).read_bytes()) not in [data["before"], data["instrumented"]]:
                raise ValueError(f"Concurrent source edit; retained backup, refusing overwrite: {source}")
        for source, data in manifest.items():
            original = (folder / data["backup"]).read_bytes()
            assert sha(original) == data["before"]
            (ROOT / source).write_bytes(original)
        (folder / "restored.json").write_text(json.dumps({source:sha((ROOT/source).read_bytes()) for source in manifest}, indent=2), encoding="utf-8")
    print("RENDER_INSTRUMENTATION_" + args.mode.upper() + "_PASS")


if __name__ == "__main__":
    main()
