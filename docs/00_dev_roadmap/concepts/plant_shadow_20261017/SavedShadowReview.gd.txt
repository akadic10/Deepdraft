extends "res://tools/performance/CameraBenchmark.gd"

const TerrainLight = preload("res://scripts/components/TerrainLighting.gd")

func _prepare_workload() -> void:
	clock_node.hour = 18.65
	root.get_node("SkyController")._update(clock_node.hour)
	root.get_node("SkyController").set_process(false)

func _set_sampling_camera() -> void:
	super._set_sampling_camera()
	var closest := INF
	var selected := Vector3i.ZERO
	for record: Dictionary in scene.get_node("SurfaceDetailManager")._records.values():
		if record.definition != "base:flora:wild_strawberry_bush" or not is_instance_valid(record.node): continue
		var at: Vector3i = record.origin
		var distance := Vector3(at).distance_to(origin)
		if distance >= closest: continue
		var beside_cliff := false
		for step in [6,9,12]:
			if generator.get_surface_y(at.x-step,at.z-step) <= at.y-4: beside_cliff = true
		if beside_cliff:
			closest = distance
			selected = at
	assert(closest<200)
	origin = Vector3(selected)+Vector3(-4,0,-4)
	rig._target_pos = origin
	rig.global_position = origin
	rig._target_zoom = 32
	print("SHADOW_REVIEW_PLANT ",selected)

func _benchmark_modes() -> Array[String]:
	return ["paused_old","paused_fixed","paused_old_repeat","paused_fixed_repeat"]

func _sample(mode: String) -> void:
	var light: DirectionalLight3D = root.get_node("SkyController")._sun
	if mode.contains("old"): light.shadow_caster_mask &= ~TerrainLight.TERRAIN_LAYER
	else: TerrainLight.configure_sun(light)
	await super._sample(mode)
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(output+mode+".png")
