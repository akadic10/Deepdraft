from pathlib import Path
import re

root=Path('P:/Deepdraft')
scratch=root/'tmp/duck_performance_20261014'
methods={
'DuckAgent':{'advance':'seconds,hours,threats','_pose':'','_try_flight':'threat,distance','_advance_flight':'seconds','_decide_duck':'here,threat,distance'},
'DuckNavigation':{'water_at':'col','surface':'col,safe','calm_water':'col,wet','clear_body':'point,radius','segment_clear':'from,to,radius,airborne','neighbors':'from','flight_path':'from,to','landing_candidates':'origin,rng,prefer_shore'},
'WildlifeManager':{'_update_flock_context':'duck','_update_herd_context':'animal','_update_wolf_target':'wolf'}}
paths={'DuckAgent':'scripts/entities','DuckNavigation':'scripts/components','WildlifeManager':'scripts/systems'}
for name,functions in methods.items():
    source=(scratch/(name+'.before.txt')).read_text(encoding='utf-8')
    source=source.replace('\n','\nconst Profile = preload("res://tmp/duck_performance_20261014/Profile.gd")\n',1)
    for method,args in functions.items():
        match=re.search(r'^func '+re.escape(method)+r'\(.*?\) -> (.+):$',source,re.M)
        assert match,(name,method)
        signature=match.group()
        ret=match.group(1)
        call='_timed_'+method+'('+args+')'
        wrapper=signature+'\n\tvar started := Time.get_ticks_usec() if Profile.enabled else 0\n'
        wrapper+=('\t'+call if ret=='void' else '\tvar result: '+ret+' = '+call)+'\n'
        wrapper+='\tProfile.record("'+name+'/'+method+'",started)\n'
        if ret!='void': wrapper+='\treturn result\n'
        source=source.replace(signature,wrapper+'\n'+signature.replace(method,'_timed_'+method),1)
    if name=='WildlifeManager':
        signature='func advance(delta: float, threats: Array) -> void:'
        source=source.replace(signature,'''func get_duck_profile() -> Dictionary:
	return {"totals":Profile.totals.duplicate(),"calls":Profile.calls.duplicate(),"peaks":Profile.peaks.duplicate()}

func advance(delta: float, threats: Array) -> void:
	Profile.reset(profile_updates)
	_timed_advance(delta, threats)
	if profile_updates: last_update_ms.merge(Profile.totals)

func _timed_advance(delta: float, threats: Array) -> void:''')
    (root/paths[name]/(name+'.gd')).write_bytes(source.encode('utf-8'))
print('Instrumented three scripts; originals retained in scratch.')
